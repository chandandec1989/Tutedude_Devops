## Set ssh-keygen in powershell administration
1. ssh-keygen -t rsa -b 4096 -C "chandan.singhdec1989@gmail.com" 
2. Set-Service ssh-agent -StartupType Automatic
3. Start-Service ssh-agent
4. ssh-add "$env:USERPROFILE\.ssh\id_rsa"
5. Get-ChildItem "$env:USERPROFILE\.ssh"
